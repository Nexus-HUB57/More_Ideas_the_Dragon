## 1. Sistema de Backtesting Histórico

### `backtesting/historical.py`
```python
import pandas as pd
from datetime import datetime, timedelta
from decimal import Decimal
from typing import List, Dict
import ccxt

class HistoricalBacktester:
    def __init__(self):
        self.exchange = ccxt.binance({'enableRateLimit': True})
    
    async def fetch_historical_data(self, pair: str, days: int = 30, timeframe: str = '1m') -> pd.DataFrame:
        """Obtém dados históricos para backtesting"""
        since = self.exchange.parse8601((datetime.now() - timedelta(days=days)).isoformat())
        ohlcv = await self.exchange.fetch_ohlcv(pair, timeframe, since)
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
        return df

    def run_backtest(self, df: pd.DataFrame, strategy: callable, initial_balance: Decimal = Decimal('1000')) -> Dict:
        """Executa backtest com uma estratégia personalizada"""
        balance = initial_balance
        position = Decimal('0')
        trades = []
        
        for i, row in df.iterrows():
            signal = strategy(row, df[:i])
            
            if signal == 'buy' and balance > 0:
                buy_amount = balance / Decimal(str(row['close']))
                position += buy_amount
                balance = Decimal('0')
                trades.append({
                    'type': 'buy',
                    'price': Decimal(str(row['close'])),
                    'amount': buy_amount,
                    'timestamp': row['timestamp']
                })
                
            elif signal == 'sell' and position > 0:
                sell_value = position * Decimal(str(row['close']))
                balance += sell_value
                position = Decimal('0')
                trades.append({
                    'type': 'sell',
                    'price': Decimal(str(row['close'])),
                    'amount': sell_value,
                    'timestamp': row['timestamp']
                })
        
        final_balance = balance + (position * Decimal(str(df.iloc[-1]['close'])))
        roi = ((final_balance - initial_balance) / initial_balance) * 100
        
        return {
            'initial_balance': initial_balance,
            'final_balance': final_balance,
            'roi': roi,
            'trades': trades,
            'sharpe_ratio': self.calculate_sharpe(df, trades)
        }

    def calculate_sharpe(self, df: pd.DataFrame, trades: List[Dict]) -> Decimal:
        """Calcula o Sharpe Ratio dos trades"""
        if not trades:
            return Decimal('0')
            
        returns = []
        prev_trade = None
        for trade in trades:
            if prev_trade and prev_trade['type'] == 'buy':
                returns.append((trade['price'] - prev_trade['price']) / prev_trade['price'])
            prev_trade = trade
        
        if not returns:
            return Decimal('0')
            
        avg_return = sum(returns) / len(returns)
        std_dev = (sum((r - avg_return)**2 for r in returns) / len(returns))**0.5
        return avg_return / std_dev if std_dev != 0 else Decimal('0')
```

## 2. Sistema de Machine Learning para Previsão

### `ml/predictor.py`
```python
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from joblib import dump, load
from decimal import Decimal
import talib

class MLPredictor:
    def __init__(self):
        self.model = RandomForestClassifier(n_estimators=100)
        self.scaler = StandardScaler()
        self.model_path = 'models/arbitrage_predictor.joblib'
    
    def prepare_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """Cria features técnicas para o modelo"""
        df['rsi'] = talib.RSI(df['close'], timeperiod=14)
        df['macd'], _, _ = talib.MACD(df['close'])
        df['bollinger_upper'], df['bollinger_middle'], df['bollinger_lower'] = talib.BBANDS(df['close'])
        df['atr'] = talib.ATR(df['high'], df['low'], df['close'], timeperiod=14)
        df['obv'] = talib.OBV(df['close'], df['volume'])
        df['returns'] = df['close'].pct_change()
        return df.dropna()

    def train_model(self, X: pd.DataFrame, y: pd.Series):
        """Treina o modelo de machine learning"""
        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled, y)
        dump((self.model, self.scaler), self.model_path)

    def predict_opportunity(self, market_data: Dict) -> Decimal:
        """Faz previsão de probabilidade de oportunidade"""
        try:
            model, scaler = load(self.model_path)
        except:
            return Decimal('0')
            
        features = pd.DataFrame([market_data])
        features = self.prepare_features(features)
        if features.empty:
            return Decimal('0')
            
        X_scaled = scaler.transform(features)
        proba = model.predict_proba(X_scaled)[0][1]
        return Decimal(str(proba))
```

## 3. Sistema de Alocação de Capital Dinâmico

### `portfolio/allocator.py`
```python
from decimal import Decimal, getcontext
from typing import Dict, List
import numpy as np
from scipy.optimize import minimize

getcontext().prec = 8

class PortfolioAllocator:
    def __init__(self):
        self.risk_free_rate = Decimal('0.0001')  # Taxa livre de risco diária
    
    def calculate_efficient_frontier(self, strategies: List[Dict]) -> Dict:
        """Calcula a fronteira eficiente para múltiplas estratégias"""
        returns = np.array([float(s['expected_return']) for s in strategies])
        risks = np.array([float(s['risk']) for s in strategies])
        corr_matrix = np.array([s['correlation'] for s in strategies])
        
        n = len(strategies)
        initial_guess = [1/n for _ in range(n)]
        bounds = [(0, 1) for _ in range(n)]
        
        # Função para minimizar o risco
        def portfolio_risk(weights):
            return np.sqrt(np.dot(weights.T, np.dot(corr_matrix * risks * risks.T, weights)))
        
        # Restrição: soma dos pesos = 1
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - 1})
        
        # Otimização
        result = minimize(
            portfolio_risk,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints
        )
        
        optimal_weights = [Decimal(str(w)) for w in result.x]
        optimal_return = sum(w * Decimal(str(r)) for w, r in zip(result.x, returns))
        
        return {
            'weights': {s['name']: w for s, w in zip(strategies, optimal_weights)},
            'expected_return': optimal_return,
            'expected_risk': Decimal(str(result.fun))
        }
    
    def dynamic_allocation(self, strategies: List[Dict], total_capital: Decimal) -> Dict:
        """Aloca capital dinamicamente entre estratégias"""
        frontier = self.calculate_efficient_frontier(strategies)
        
        allocation = {}
        for name, weight in frontier['weights'].items():
            allocation[name] = {
                'capital': total_capital * weight,
                'weight': weight
            }
        
        return {
            'allocation': allocation,
            'portfolio_return': frontier['expected_return'],
            'portfolio_risk': frontier['expected_risk'],
            'sharpe_ratio': (frontier['expected_return'] - self.risk_free_rate) / frontier['expected_risk']
        }
```

## 4. Sistema de Relatórios Automatizados

### `reporting/generator.py`
```python
import pandas as pd
from datetime import datetime
from decimal import Decimal
from typing import Dict, List
import matplotlib.pyplot as plt
import seaborn as sns
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.image import MIMEImage
import smtplib
import os

class ReportGenerator:
    def __init__(self):
        self.output_dir = 'reports'
        os.makedirs(self.output_dir, exist_ok=True)
    
    def generate_daily_report(self, trades: List[Dict], performance: Dict) -> str:
        """Gera relatório diário em HTML e PDF"""
        df = pd.DataFrame(trades)
        df['profit'] = df.apply(lambda x: x['amount'] * (x['sell_price'] - x['buy_price']) if x['type'] == 'arbitrage' else 0, axis=1)
        
        # Cria visualizações
        self._create_performance_chart(performance)
        self._create_trades_chart(df)
        
        # Gera HTML
        html = f"""
        <html>
            <body>
                <h1>Relatório Diário - {datetime.today().strftime('%d/%m/%Y')}</h1>
                <h2>Performance</h2>
                <p>ROI: {performance['roi']:.2f}%</p>
                <p>Sharpe Ratio: {performance['sharpe_ratio']:.2f}</p>
                <img src="performance.png" alt="Performance">
                
                <h2>Trades</h2>
                <p>Total de trades: {len(df)}</p>
                <p>Lucro total: ${performance['total_profit']:.2f}</p>
                <img src="trades.png" alt="Trades">
            </body>
        </html>
        """
        
        # Salva arquivos
        report_path = os.path.join(self.output_dir, f"report_{datetime.today().strftime('%Y%m%d')}.html")
        with open(report_path, 'w') as f:
            f.write(html)
            
        return report_path
    
    def _create_performance_chart(self, performance: Dict):
        plt.figure(figsize=(10, 5))
        sns.lineplot(x=performance['history']['date'], y=performance['history']['balance'])
        plt.title('Evolução do Saldo')
        plt.savefig(os.path.join(self.output_dir, 'performance.png'))
        plt.close()
    
    def _create_trades_chart(self, df: pd.DataFrame):
        plt.figure(figsize=(10, 5))
        df['cum_profit'] = df['profit'].cumsum()
        sns.lineplot(x=df['timestamp'], y=df['cum_profit'])
        plt.title('Lucro Acumulado')
        plt.savefig(os.path.join(self.output_dir, 'trades.png'))
        plt.close()
    
    def send_email_report(self, to_email: str, report_path: str):
        """Envia relatório por email"""
        msg = MIMEMultipart()
        msg['Subject'] = f"Relatório Diário - {datetime.today().strftime('%d/%m/%Y')}"
        msg['From'] = 'bot@arbitrage.com'
        msg['To'] = to_email
        
        # Corpo do email
        with open(report_path, 'r') as f:
            html = f.read()
        msg.attach(MIMEText(html, 'html'))
        
        # Anexa imagens
        for img in ['performance.png', 'trades.png']:
            with open(os.path.join(self.output_dir, img), 'rb') as f:
                img_data = f.read()
            image = MIMEImage(img_data, name=img)
            msg.attach(image)
        
        # Envia email
        with smtplib.SMTP('smtp.gmail.com', 587) as server:
            server.starttls()
            server.login('seuemail@gmail.com', 'suasenha')
            server.send_message(msg)
```
