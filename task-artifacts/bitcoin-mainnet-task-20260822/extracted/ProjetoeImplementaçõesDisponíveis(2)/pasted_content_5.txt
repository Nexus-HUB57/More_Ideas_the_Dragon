## 1. Execução Automática de Ordens

### `trading/executor.py`
```python
import asyncio
from decimal import Decimal
from typing import Dict
from exchanges import BinanceAPI, BittrexAPI, FoxbitAPI, MercadoBitcoinAPI

class TradeExecutor:
    def __init__(self):
        self.exchanges = {
            'binance': BinanceAPI(),
            'bittrex': BittrexAPI(),
            'foxbit': FoxbitAPI(),
            'mercado_bitcoin': MercadoBitcoinAPI()
        }
        self.min_profit = Decimal('0.5')  # 0.5% de lucro mínimo
        self.max_slippage = Decimal('0.2')  # 0.2% de slippage máximo

    async def execute_arbitrage(self, opportunity: Dict):
        buy_ex = self.exchanges[opportunity['buy_exchange']]
        sell_ex = self.exchanges[opportunity['sell_exchange']]
        pair = opportunity['pair']
        amount = self._calculate_amount(opportunity)

        try:
            # Executa a ordem de compra
            buy_order = await buy_ex.create_order(
                symbol=pair,
                type='limit',
                side='buy',
                amount=amount,
                price=opportunity['buy_price']
            )

            # Executa a ordem de venda
            sell_order = await sell_ex.create_order(
                symbol=pair,
                type='limit',
                side='sell',
                amount=amount,
                price=opportunity['sell_price']
            )

            return {
                'status': 'success',
                'buy_order': buy_order,
                'sell_order': sell_order,
                'profit': (sell_order['filled'] * sell_order['price']) - 
                         (buy_order['filled'] * buy_order['price'])
            }

        except Exception as e:
            return {
                'status': 'error',
                'error': str(e),
                'pair': pair,
                'buy_exchange': opportunity['buy_exchange'],
                'sell_exchange': opportunity['sell_exchange']
            }

    def _calculate_amount(self, opportunity: Dict) -> Decimal:
        """Calcula o tamanho da posição baseado no saldo disponível"""
        exchange = self.exchanges[opportunity['buy_exchange']]
        balance = exchange.get_balance()
        base_currency = opportunity['pair'].split('/')[0]
        
        available = balance['free'].get(base_currency, Decimal('0'))
        max_amount = min(
            available,
            Decimal('0.1')  # Limite para evitar risco excessivo
        )
        return max_amount
```

## 2. Cálculo de Fees no Spread

### `arbitrage/calculator.py`
```python
from decimal import Decimal
from typing import Dict

class ProfitCalculator:
    @staticmethod
    def calculate_net_profit(opportunity: Dict, fees: Dict) -> Decimal:
        """
        Calcula o lucro líquido considerando:
        - Fees de compra e venda
        - Slippage estimado
        - Custos de transferência (se aplicável)
        """
        buy_fee = opportunity['amount'] * opportunity['buy_price'] * fees[opportunity['buy_exchange']]['taker']
        sell_fee = opportunity['amount'] * opportunity['sell_price'] * fees[opportunity['sell_exchange']]['taker']
        
        gross_profit = (opportunity['sell_price'] - opportunity['buy_price']) * opportunity['amount']
        net_profit = gross_profit - buy_fee - sell_fee
        
        # Aplica slippage estimado
        net_profit *= (1 - opportunity['slippage'])
        
        return net_profit

    @staticmethod
    def get_exchange_fees() -> Dict:
        """Retorna a estrutura de fees para cada exchange"""
        return {
            'binance': {
                'taker': Decimal('0.001'),
                'maker': Decimal('0.001'),
                'withdrawal': {
                    'BTC': Decimal('0.0005')
                }
```

## 3. Implementação de WebSocket

### `websocket/client.py`
```python
import asyncio
import websockets
import json
from typing import Dict, Callable

class WebSocketClient:
    def __init__(self, uri: str):
        self.uri = uri
        self.websocket = None
        self.subscriptions = set()
        self.message_handler = None

    async def connect(self):
        self.websocket = await websockets.connect(self.uri)
        return self.websocket

    async def subscribe(self, channels: list):
        if self.websocket:
            subscription_msg = {
                "method": "SUBSCRIBE",
                "params": channels,
                "id": 1
            }
            await self.websocket.send(json.dumps(subscription_msg))
            self.subscriptions.update(channels)

    async def listen(self, handler: Callable):
        self.message_handler = handler
        while True:
            try:
                message = await self.websocket.recv()
                data = json.loads(message)
                await self.message_handler(data)
            except Exception as e:
                print(f"WebSocket error: {e}")
                await self.reconnect()

    async def reconnect(self):
        await asyncio.sleep(5)
        self.websocket = await self.connect()
        if self.subscriptions:
            await self.subscribe(list(self.subscriptions)))

    async def close(self):
        await self.websocket.close()
```

### `exchanges/binance_websocket.py`
```python
from websocket.client import WebSocketClient
from decimal import Decimal
import asyncio

class BinanceWebSocket:
    def __init__(self):
        self.uri = "wss://stream.binance.com:9443/ws"
        self.client = WebSocketClient(self.uri)
        self.order_books = {}

    async def handle_book_ticker(self, data: Dict):
        symbol = data['s']
        bid_price = Decimal(data['b'])
        ask_price = Decimal(data['a'])
        
        self.order_books[symbol] = {
            'bid': bid_price,
            'ask': ask_price,
            'timestamp': data['E']
        }

    async def start(self, symbols: list):
        await self.client.connect()
        channels = [f"{symbol.lower()}@bookTicker" for symbol in symbols]
        await self.client.subscribe(channels)
        await self.client.listen(self.handle_book_ticker)
```

## 4. Monitoramento de Saldo em Tempo Real

### `balance/monitor.py`
```python
from decimal import Decimal
from typing import Dict
import asyncio
from exchanges import BinanceAPI, BittrexAPI, FoxbitAPI, MercadoBitcoinAPI

class BalanceMonitor:
    def __init__(self):
        self.exchanges = {
            'binance': BinanceAPI(),
            'bittrex': BittrexAPI(),
            'foxbit': FoxbitAPI(),
            'mercado_bitcoin': MercadoBitcoinAPI()
        }
        self.balances = {}
        self.thresholds = {
            'BTC': Decimal('0.01'),
            'USDT': Decimal('100'),
            'BRL': Decimal('500')
        }

    async def monitor_balances(self):
        while True:
            for exchange_name, exchange in self.exchanges.items():
                try:
                    balance = exchange.get_balance()
                    self.balances[exchange_name] = balance
                    
                    # Verifica se algum saldo está abaixo do threshold
                    for currency, amount in balance['free'].items():
                        if currency in self.thresholds and amount < self.thresholds[currency]:
                            print(f"ALERTA: Saldo de {currency} na {exchange_name} está abaixo do limite!")
                    
                except Exception as e:
                    print(f"Erro ao verificar saldo na {exchange_name}: {e}")
            
            await asyncio.sleep(60)  # Verifica a cada minuto

    def get_available_balance(self, exchange: str, currency: str) -> Decimal:
        return self.balances.get(exchange, {}).get('free', {}).get(currency, Decimal('0'))
```

## 5. Integração com o Dashboard

### `dashboard/integration.py`
```python
from fastapi import FastAPI, WebSocket
from fastapi.middleware.cors import CORSMiddleware
import uvicorn
from arbitrage.monitor import ArbitrageMonitor
from balance.monitor import BalanceMonitor
import asyncio
import json

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

monitor = ArbitrageMonitor()
balance_monitor = BalanceMonitor()

@app.websocket("/ws/arbitrage")
async def arbitrage_websocket(websocket: WebSocket):
    await websocket.accept()
    async for opportunity in monitor.monitor_pair('BTC/USDT'):
        await websocket.send_json(opportunity)

@app.websocket("/ws/balance")
async def balance_websocket(websocket: WebSocket):
    await websocket.accept()
    while True:
        balances = balance_monitor.balances
        await websocket.send_json(balances)
        await asyncio.sleep(10)

async def run_monitors():
    await asyncio.gather(
        monitor.monitor_all_pairs(),
        balance_monitor.monitor_balances()
    )

@app.on_event("startup")
async def startup_event():
    asyncio.create_task(run_monitors())

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
